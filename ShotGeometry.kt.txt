package com.poolpredictor2.geometry

import kotlin.math.hypot
import kotlin.math.sqrt

// Project 2 - Android from scratch
// LOCK: 2 garis doang, no Auto Play
// 1. Cue ball -> titik tumbukan (ghost)
// 2. Target ball -> pocket (atau titik berhenti kalo power kurang)

data class Point(val x: Float, val y: Float)
data class Rect(val left: Float, val top: Float, val right: Float, val bottom: Float)
data class Ball(val number: Int, val pos: Point, val radius: Float)
data class Line(val start: Point, val end: Point, val colorHex: String)

data class ShotPrediction(
    val cueLine: Line?,       // garis 1
    val targetLine: Line?,    // garis 2
    val ghostBall: Point,     // titik tumbukan
    val willReachPocket: Boolean
)

object ShotGeometry {

    // Warna bola - harus sama dengan themes.json -> ballColors
    fun getBallColor(number: Int): String {
        return when (number) {
            1, 9 -> "#FFD600"  // kuning
            2, 10 -> "#2196F3" // biru
            3, 11 -> "#F44336" // merah
            4, 12 -> "#9C27B0" // ungu
            5, 13 -> "#FF9800" // oranye
            6, 14 -> "#4CAF50" // hijau
            7, 15 -> "#795548" // maroon
            8 -> "#111111"     // hitam
            else -> "#FFFFFF"   // cue ball = putih
        }
    }

    /**
     * Hitung 2 garis, power-aware.
     * @param powerRatio 0.0 - 1.0, dari power cue. 0 = tidak ditarik, 1 = full power
     * @param frictionFactor jarak yang bisa ditempuh di power 1.0, buat simulasi kasar gesekan
     */
    fun calculateShot(
        cue: Ball,
        target: Ball,
        pocket: Point,
        powerRatio: Float,
        table: Rect,
        frictionFactor: Float = 1500f
    ): ShotPrediction {
        val power = powerRatio.coerceIn(0f, 1f)
        if (power <= 0.01f) {
            return ShotPrediction(null, null, target.pos, false)
        }

        // 1. Ghost ball = titik kontak: target - arah ke pocket * (2 * radius)
        val dxPocket = pocket.x - target.pos.x
        val dyPocket = pocket.y - target.pos.y
        val distPocket = hypot(dxPocket, dyPocket).coerceAtLeast(0.001f)
        val ux = dxPocket / distPocket
        val uy = dyPocket / distPocket

        val ghost = Point(
            target.pos.x - ux * (target.radius * 2f),
            target.pos.y - uy * (target.radius * 2f)
        )

        // 2. Jarak yang bisa ditempuh dengan power ini (kasar, buat lawan garis full-power asal)
        val maxTravel = power * frictionFactor

        // Jarak cue -> ghost
        val distCueToGhost = hypot(ghost.x - cue.pos.x, ghost.y - cue.pos.y)

        // Kalo power tidak cukup sampe ghost, garis cue berhenti di tengah, target tidak jalan
        if (maxTravel < distCueToGhost) {
            val t = maxTravel / distCueToGhost.coerceAtLeast(0.001f)
            val stop = Point(
                cue.pos.x + (ghost.x - cue.pos.x) * t,
                cue.pos.y + (ghost.y - cue.pos.y) * t
            )
            val cueLine = Line(
                clipInside(cue.pos, table),
                clipInside(stop, table),
                "#FFFFFF" // cue = putih
            )
            return ShotPrediction(cueLine, null, ghost, false)
        }

        // Garis 1: cue -> ghost, di-clip di dalam meja
        val cueLine = Line(
            clipInside(cue.pos, table),
            clipInside(ghost, table),
            "#FFFFFF"
        )

        // Sisa power buat target ball
        val remaining = maxTravel - distCueToGhost
        val targetColor = getBallColor(target.number)

        // Kalo sisa power tidak cukup sampe pocket, garis target berhenti di tengah
        // Ini yang bikin beda dari screenshot: garisnya tidak dipaksa sampe pocket
        val targetEnd = if (remaining < distPocket) {
            val t = remaining / distPocket.coerceAtLeast(0.001f)
            Point(
                target.pos.x + (pocket.x - target.pos.x) * t,
                target.pos.y + (pocket.y - target.pos.y) * t
            )
        } else {
            pocket
        }

        val targetLine = Line(
            clipInside(target.pos, table),
            clipInside(targetEnd, table),
            targetColor
        )

        return ShotPrediction(
            cueLine = cueLine,
            targetLine = targetLine,
            ghostBall = ghost,
            willReachPocket = remaining >= distPocket
        )
    }

    // Clip titik di dalam meja, biar garis tidak keluar tata surya
    private fun clipInside(p: Point, table: Rect): Point {
        return Point(
            p.x.coerceIn(table.left, table.right),
            p.y.coerceIn(table.top, table.bottom)
        )
    }
}
